## Import

```ts
import { TraceDetailView } from '@traces/react'
import type { Trace } from '@traces/react'
```

## Props

| Prop | Type | Required | Default | Description |
|---|---|---|---|---|
| `trace` | `Trace` | yes | — | The trace to display |
| `selectedSpanId` | `string \| null` | yes | — | Currently selected span; controls `SpanDetailPanel` open state |
| `onSelectSpan` | `(id: string \| null) => void` | yes | — | Called when selection changes (span click, keyboard, panel close) |
| `deepLinkSpanId` | `string \| null` | no | — | Select and scroll to this span on mount — pass from URL param |

## Usage

```tsx
import { useState } from 'react'
import { TraceDetailView } from '@traces/react'

function TracePage({ trace }) {
  const [selectedSpanId, setSelectedSpanId] = useState<string | null>(null)

  return (
    <TraceDetailView
      trace={trace}
      selectedSpanId={selectedSpanId}
      onSelectSpan={setSelectedSpanId}
      deepLinkSpanId={new URLSearchParams(location.search).get('span')}
    />
  )
}
```

## Behavior

- composes `TraceWaterfall` + `SpanDetailPanel` — manages the panel open/close state from `selectedSpanId`
- keyboard navigation is active on the waterfall: `↑`/`↓` move, `←`/`→` collapse/expand, `Esc` deselects and closes the panel
- `deepLinkSpanId` triggers `scrollToSpanId` on the waterfall so the target row is visible without user interaction
- header shows trace ID, root service, total span count, error count, and overall duration
